          and identical configuration flexibility at every step.

CRITICAL RULE: After each phase, verify that autotools and CMake produce
               identical binaries and identical configuration behavior.

   file "$CMAKE_BIN"
   
   echo "=== Symbol Count ==="

   nm -g "$CMAKE_BIN" | wc -l
   
   echo "=== Symbol Diff ==="
   nm -g "$AUTOTOOLS_BIN" | sort > /tmp/sym-auto.txt
   nm -g "$CMAKE_BIN" | sort > /tmp/sym-cmake.txt
   diff /tmp/sym-auto.txt /tmp/sym-cmake.txt
   
   echo "=== Linked Libraries ==="
   otool -L "$AUTOTOOLS_BIN"
   otool -L "$CMAKE_BIN"
   
   echo "=== Embedded Strings Sample ==="
   strings "$AUTOTOOLS_BIN" | grep -v "^/Users" | sort > /tmp/str-auto.txt
